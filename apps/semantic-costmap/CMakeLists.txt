cmake_minimum_required(VERSION 3.16)
project(demo_semantic_costmap)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep nav2_costmap_2d nav2_util pluginlib rclcpp sensor_msgs std_msgs std_srvs tf2_ros tf2_geometry_msgs)
  find_package(${dep} REQUIRED)
endforeach()
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(${PROJECT_NAME} SHARED src/semantic_layer.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} nav2_costmap_2d nav2_util pluginlib rclcpp sensor_msgs std_msgs tf2_ros tf2_geometry_msgs)
target_link_libraries(${PROJECT_NAME} nlohmann_json::nlohmann_json OpenSSL::Crypto)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
add_executable(phase4_observer src/phase4_observer.cpp)
target_link_libraries(phase4_observer ${PROJECT_NAME})
ament_target_dependencies(phase4_observer nav2_costmap_2d nav2_util pluginlib rclcpp std_srvs tf2_ros)
install(TARGETS phase4_observer DESTINATION lib/${PROJECT_NAME})
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_semantic_layer test/test_semantic_layer.cpp)
  target_link_libraries(test_semantic_layer ${PROJECT_NAME})
  ament_target_dependencies(test_semantic_layer nav2_costmap_2d nav2_util pluginlib rclcpp sensor_msgs std_msgs tf2_ros)
endif()
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(nav2_costmap_2d nav2_util rclcpp sensor_msgs std_msgs tf2_ros tf2_geometry_msgs nlohmann_json OpenSSL)
ament_package()
