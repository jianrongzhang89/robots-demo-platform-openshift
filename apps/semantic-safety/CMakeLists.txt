cmake_minimum_required(VERSION 3.16)
project(phase5_drive_watchdog)
set(CMAKE_CXX_STANDARD 17)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
find_package(GTest REQUIRED)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(CycloneDDS REQUIRED)
add_library(phase5_drive_watchdog SHARED native/watchdog.cpp)
target_link_libraries(phase5_drive_watchdog gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13)
target_compile_options(phase5_drive_watchdog PRIVATE -Wall -Wextra -Werror)
add_executable(test_native_lease native/test_lease.cpp)
target_link_libraries(test_native_lease GTest::gtest_main)
enable_testing()
add_test(NAME native_lease COMMAND test_native_lease)
install(TARGETS phase5_drive_watchdog LIBRARY DESTINATION lib)
add_executable(phase5_command_observer native/command_observer.cpp)
ament_target_dependencies(phase5_command_observer rclcpp geometry_msgs std_msgs)
target_link_libraries(phase5_command_observer CycloneDDS::ddsc)
target_compile_options(phase5_command_observer PRIVATE -Wall -Wextra -Werror)
install(TARGETS phase5_command_observer RUNTIME DESTINATION bin)
add_library(phase5_contact_audit SHARED native/contact_audit.cpp)
target_link_libraries(phase5_contact_audit gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13)
target_compile_options(phase5_contact_audit PRIVATE -Wall -Wextra -Werror)
install(TARGETS phase5_contact_audit LIBRARY DESTINATION lib)
